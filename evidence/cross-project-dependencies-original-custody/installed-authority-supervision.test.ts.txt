function exists(pid: number): boolean {
  try {
    process.kill(pid, 0);
    return true;
  } catch (error) {
    if (error instanceof Error && 'code' in error && error.code === 'ESRCH') return false;
    throw error;
  }
}

interface AuthorityFixtureCase {
  key: string;
  name: string;
  expectedFailure?: string;
}

const STRUCTURAL_RUNTIME_PIN = 'PHASE1_UNREVIEWED_NO_RUNTIME_AUTHORIZATION';
const PREPARATION_DIRECTORY =
  '/__REDACTED_PRIVATE_CUSTODY_PREPARATION__';

async function readAuthorityPreparation(scenario: string, inputs: unknown) {
  if (!/^[a-f0-9]{64}$/.test(STRUCTURAL_RUNTIME_PIN))
    throw new Error('reviewed structural runtime pin is absent');
  const pinBytes = await readFile(PREPARATION_DIRECTORY + '/runtime-authorization.json');
  if (createHash('sha256').update(pinBytes).digest('hex') !== STRUCTURAL_RUNTIME_PIN)
    throw new Error('structural runtime authorization bytes differ');
  const decoded: unknown = JSON.parse(pinBytes.toString('utf8'));
  const pin = readClosed(decoded, [
    'version',
    'bundleSha256',
    'driverSha256',
    'programsSha256',
    'exports',
  ]);
  if (
    pin['version'] !== 1 ||
    !Array.isArray(pin['exports']) ||
    !pin['exports'].every((member) => typeof member === 'string')
  )
    throw new Error('structural runtime authorization shape differs');
  const readPinned = async (name: string, expected: unknown) => {
    if (typeof expected !== 'string' || !/^[a-f0-9]{64}$/.test(expected))
      throw new Error('structural artifact pin malformed');
    const bytes = await readFile(PREPARATION_DIRECTORY + '/' + name);
    if (createHash('sha256').update(bytes).digest('hex') !== expected)
      throw new Error('structural artifact bytes differ');
    return bytes;
  };
  await readPinned('candidate.mjs', pin['bundleSha256']);
  const driver = await readPinned('driver-template.mjs', pin['driverSha256']);
  const programBytes = await readPinned('programs.json', pin['programsSha256']);
  const programs: unknown = JSON.parse(programBytes.toString('utf8'));
  if (
    typeof programs !== 'object' ||
    programs === null ||
    Array.isArray(programs) ||
    !(scenario in programs)
  )
    throw new Error('reviewed fixture scenario missing');
  const program: unknown = Reflect.get(programs, scenario);
  if (typeof program !== 'string') throw new Error('reviewed fixture program malformed');
  const programSha256 = createHash('sha256').update(program).digest('hex');
  const configuration = {
    scenario,
    inputs,
    program,
    programSha256,
    bundleSha256: pin['bundleSha256'],
    exports: pin['exports'],
  };
  return {
    driver:
      'const fixtureConfiguration=' +
      JSON.stringify(configuration) +
      ';\n' +
      driver.toString('utf8'),
    programSha256,
    executable:
      scenario === 'async-spawn' ? '/nonexistent/owned-original-executable' : '/usr/bin/python3',
    argumentsSha256: createHash('sha256')
      .update(JSON.stringify(scenario === 'async-spawn' ? [] : ['-I', '-c', program]))
      .digest('hex'),
  };
}

/** Runs only the fixed Bun driver. No original factory executes in the test runner.
 * The parent retains independent outer exit/error/EOF/input/close observations before
 * parsing or asserting. It sends retirement only after an attributable fixture ACK.
 * Failure to obtain actual closure within the authored bounds enters visible HOLD;
 * it never kills an unacknowledged driver or treats a timeout as successful cleanup.
 */
async function runAuthorityFixture(scenario: AuthorityFixtureCase): Promise<void> {
  const inputs = {
    ...(await request()),
    databases: [
      { family: 'prod', path: '/fixture/prod.sqlite' },
      { family: 'dev', path: '/fixture/dev.sqlite' },
    ],
  };
  const prepared = await readAuthorityPreparation(scenario.key, inputs);
  const outer = spawn(process.execPath, ['-e', prepared.driver], {
    stdio: ['pipe', 'pipe', 'pipe'],
  });
  let releaseOwner: () => void = () => {
    throw new Error('outer fixture owner missing');
  };
  const ownership = new Promise<void>((resolve) => {
    releaseOwner = resolve;
  });
  authorityFixtureOwners.add(ownership);
  const facts: {
    pid: number | undefined;
    exit: { code: number | null; signal: NodeJS.Signals | null } | undefined;
    errors: string[];
    closed: boolean;
    stdoutEof: boolean;
    stderrEof: boolean;
    inputFinished: boolean;
    inputClosed: boolean;
  } = {
    pid: outer.pid,
    exit: undefined,
    errors: [],
    closed: false,
    stdoutEof: false,
    stderrEof: false,
    inputFinished: false,
    inputClosed: false,
  };
  const exited = new Promise<void>((resolve) => {
    outer.on('exit', (code, signal) => {
      facts.exit = { code, signal };
      resolve();
    });
  });
  const closed = new Promise<void>((resolve) => {
    outer.on('close', () => {
      facts.closed = true;
      resolve();
    });
  });
  outer.on('error', (cause) => {
    facts.errors.push(cause.name);
  });
  outer.stdin.on('error', (cause) => {
    facts.errors.push(cause.name);
  });
  outer.stdin.on('finish', () => {
    facts.inputFinished = true;
  });
  outer.stdin.on('close', () => {
    facts.inputClosed = true;
  });
  let retire: (frame: Record<string, unknown>) => void = () => {
    throw new Error('fixture ACK observer missing');
  };
  const retired = new Promise<Record<string, unknown>>((resolve) => {
    retire = resolve;
  });
  let parentFailure: unknown;
  let finalDriverFrame: Record<string, unknown> | undefined;
  const observed = { bindingTrapped: false, driverRetiring: false };
  let stderrBytes = 0;
  const stdout = (async () => {
    let buffered = '';
    let bytes = 0;
    for await (const chunk of outer.stdout) {
      if (!Buffer.isBuffer(chunk)) throw new Error('fixture driver emitted nonbytes');
      bytes += chunk.length;
      if (bytes > 131072) {
        parentFailure = new Error('fixture driver output bound');
        continue;
      }
      buffered += chunk.toString('utf8');
      for (;;) {
        const newline = buffered.indexOf('\n');
        if (newline < 0) break;
        const line = buffered.slice(0, newline);
        buffered = buffered.slice(newline + 1);
        try {
          const decoded: unknown = JSON.parse(line);
          if (typeof decoded !== 'object' || decoded === null || Array.isArray(decoded))
            throw new Error('fixture frame malformed');
          const frame = readClosed(decoded, [
            'version',
            'phase',
            'scenario',
            'driverPid',
            'programSha256',
            'dispatches',
            'child',
            'failure',
            'unexpectedFailures',
            'expectedStop',
            'productionOutcome',
            'sealWitness',
            'productionCloseReturned',
            'retirementRequested',
            'fixtureLaunches',
            'ownerSealed',
            'bodySettled',
            'effectTasks',
            'uninstalledDenied',
            'deadline',
            'fixtureSpawnThrow',
          ]);
          if (frame['version'] !== 2) throw new Error('fixture ACK version differs');
          if (frame['scenario'] !== scenario.key || frame['driverPid'] !== outer.pid)
            throw new Error('fixture frame attribution differs');
          if (
            !Array.isArray(frame['unexpectedFailures']) ||
            frame['unexpectedFailures'].length !== 0 ||
            frame['failure'] !== null
          )
            parentFailure = new Error('unexpected fixture harness failure');
          if (frame['phase'] === 'binding-trapped') observed.bindingTrapped = true;
          else if (frame['phase'] === 'fixture-retired') retire(frame);
          else if (frame['phase'] === 'driver-retiring') {
            observed.driverRetiring = true;
            finalDriverFrame = frame;
          } else if (frame['phase'] === 'fixture-unknown')
            parentFailure = new Error('fixture original resources remain unknown');
          else throw new Error('fixture phase differs');
        } catch (cause) {
          parentFailure = cause;
        }
      }
    }
    facts.stdoutEof = outer.stdout.readableEnded;
    if (buffered !== '') parentFailure = new Error('fixture driver partial frame');
  })().catch((cause: unknown) => {
    parentFailure = cause;
  });
  const stderr = (async () => {
    for await (const chunk of outer.stderr) {
      if (!Buffer.isBuffer(chunk)) throw new Error('fixture driver emitted nonbyte stderr');
      stderrBytes += chunk.length;
    }
    facts.stderrEof = outer.stderr.readableEnded;
  })().catch((cause: unknown) => {
    parentFailure = cause;
  });
  const wait = async <T>(promise: Promise<T>, milliseconds: number): Promise<T> => {
    let timer: ReturnType<typeof setTimeout> | undefined;
    try {
      return await Promise.race([
        promise,
        new Promise<never>((_resolve, reject) => {
          timer = setTimeout(() => {
            reject(new Error('fixture observation bound'));
          }, milliseconds);
        }),
      ]);
    } finally {
      clearTimeout(timer);
    }
  };
  const send = (command: 'abort' | 'retire'): Promise<void> =>
    new Promise((resolve, reject) => {
      let callback = false;
      let drained = false;
      let finished = false;
      const complete = () => {
        if (callback && drained && !finished) {
          finished = true;
          resolve();
        }
      };
      const accepted = outer.stdin.write(command + '\n', (cause) => {
        if (cause) {
          reject(cause);
          return;
        }
        callback = true;
        complete();
      });
      if (accepted) {
        drained = true;
        complete();
      } else
        outer.stdin.once('drain', () => {
          drained = true;
          complete();
        });
    });
  const hold = async (reason: string): Promise<never> => {
    console.error(
      JSON.stringify({
        fixture: 'owned-original-HOLD',
        scenario: scenario.key,
        reason,
        outer: facts,
      }),
    );
    setInterval(() => undefined, 60000);
    return await new Promise<never>(() => undefined);
  };
  let frame: Record<string, unknown> | undefined;
  try {
    try {
      frame = await wait(retired, 10000);
    } catch (cause) {
      parentFailure = cause;
      try {
        await wait(send('abort'), 500);
      } catch (controlCause) {
        parentFailure = controlCause;
      }
      try {
        frame = await wait(retired, 3000);
      } catch {
        await hold('missing-attributable-fixture-teardown');
      }
    }
  } finally {
    const acknowledged = frame ?? (await hold('fixture-ACK-unavailable'));
    const expectedProgram = prepared.programSha256;
    if (acknowledged['programSha256'] !== expectedProgram)
      await hold('fixture-program-attribution-differs');
    if (!observed.bindingTrapped && scenario.key !== 'uninstalled')
      parentFailure = new Error('imported binding trap was not observed');
    if (
      acknowledged['bodySettled'] !== true ||
      acknowledged['effectTasks'] !== 0 ||
      acknowledged['ownerSealed'] !== true
    )
      await hold('original-driver-body-or-effect-work-unsettled');
    const dispatches = acknowledged['dispatches'];
    const direct = ['registry', 'driver-setup', 'seal-launch', 'seal-write'].includes(scenario.key);
    const noFixture = [
      'uninstalled',
      'binding-trap',
      'guards',
      'synchronous-spawn',
      'seal-launch',
    ].includes(scenario.key);
    const expectedDispatches =
      scenario.key === 'uninstalled'
        ? 0
        : scenario.key === 'binding-trap' || scenario.key === 'guards' || direct
          ? 1
          : 2;
    if (
      !Array.isArray(dispatches) ||
      dispatches.length !== expectedDispatches ||
      acknowledged['fixtureLaunches'] !== (noFixture ? 0 : 1)
    )
      parentFailure = new Error('case-specific dispatch or fixture count differs');
    if (scenario.key === 'uninstalled' && acknowledged['uninstalledDenied'] !== true)
      parentFailure = new Error('uninstalled default deny was not observed');
    const expectedPhases =
      scenario.key === 'uninstalled'
        ? []
        : scenario.key === 'synchronous-spawn'
          ? ['trap', 'spawn-throw']
          : expectedDispatches === 1
            ? ['trap']
            : ['trap', 'fixture'];
    if (
      Array.isArray(dispatches) &&
      dispatches.some(
        (dispatch: unknown, index: number) =>
          typeof dispatch !== 'object' ||
          dispatch === null ||
          Array.isArray(dispatch) ||
          Object.keys(dispatch).sort().join(',') !== 'phase,shapeMatches' ||
          !('phase' in dispatch) ||
          dispatch.phase !== expectedPhases[index] ||
          !('shapeMatches' in dispatch) ||
          dispatch.shapeMatches !== true,
      )
    )
      parentFailure = new Error('closed dispatch record shape or phase differs');
    const actualLaunches = acknowledged['fixtureLaunches'];
    if (
      typeof actualLaunches !== 'number' ||
      !Number.isInteger(actualLaunches) ||
      actualLaunches < 0 ||
      actualLaunches > 1 ||
      (actualLaunches === 0) !== (acknowledged['child'] === null)
    )
      await hold('actual-fixture-owner-budget-or-child-association-differs');
    const child = acknowledged['child'];
    if (child !== null) {
      if (typeof child !== 'object' || Array.isArray(child))
        await hold('fixture-child-facts-malformed');
      const retained = child as Record<string, unknown>;
      const pipeClosed = (name: string) => {
        const pipe = retained[name];
        return (
          typeof pipe === 'object' && pipe !== null && 'closed' in pipe && pipe.closed === true
        );
      };
      const input = retained['stdin'];
      const callbacksKnown =
        typeof input === 'object' &&
        input !== null &&
        'callbacks' in input &&
        'writes' in input &&
        input.callbacks === input.writes;
      const actualExit = retained['exit'];
      const actualExitKnown =
        typeof actualExit === 'object' &&
        actualExit !== null &&
        (('code' in actualExit && typeof actualExit.code === 'number') ||
          ('signal' in actualExit && typeof actualExit.signal === 'string'));
      if (retained['programSha256'] !== expectedProgram)
        await hold('fixture-child-program-attribution-differs');
      const errors = retained['errors'];
      const error =
        Array.isArray(errors) && errors.length === 1 ? (errors[0] as unknown) : undefined;
      const exactSpawnError =
        typeof error === 'object' &&
        error !== null &&
        !Array.isArray(error) &&
        Object.keys(error).sort().join(',') ===
          'argumentCount,argumentsSha256,code,name,path,syscall' &&
        'name' in error &&
        error.name === 'Error' &&
        'code' in error &&
        error.code === 'ENOENT' &&
        'path' in error &&
        error.path === prepared.executable &&
        'syscall' in error &&
        error.syscall === 'spawn ' + prepared.executable &&
        'argumentCount' in error &&
        error.argumentCount === (scenario.key === 'async-spawn' ? 0 : 3) &&
        'argumentsSha256' in error &&
        error.argumentsSha256 === prepared.argumentsSha256;
      const failedLaunch =
        retained['pid'] === null &&
        retained['spawned'] === false &&
        retained['exit'] === null &&
        exactSpawnError;
      if (retained['executable'] !== prepared.executable)
        await hold('fixture-executable-attribution-differs');
      const failedIo = (name: string) => {
        const pipe = retained[name];
        return (
          typeof pipe === 'object' &&
          pipe !== null &&
          'closed' in pipe &&
          pipe.closed === true &&
          'bytes' in pipe &&
          pipe.bytes === 0 &&
          'errors' in pipe &&
          Array.isArray(pipe.errors) &&
          pipe.errors.length === 0
        );
      };
      if (scenario.key === 'async-spawn') {
        if (
          !failedLaunch ||
          !failedIo('stdout') ||
          !failedIo('stderr') ||
          typeof input !== 'object' ||
          input === null ||
          !('writes' in input) ||
          input.writes !== 0 ||
          !('callbacks' in input) ||
          input.callbacks !== 0
        )
          parentFailure = new Error('expected ENOENT failed-launch witness differs');
      } else if (failedLaunch) parentFailure = new Error('unexpected fixture launch failure');

      if (
        (!actualExitKnown && !failedLaunch) ||
        retained['closed'] !== true ||
        !pipeClosed('stdout') ||
        !pipeClosed('stderr') ||
        !pipeClosed('stdin') ||
        !callbacksKnown
      )
        await hold('fixture-ACK-lacks-actual-closure');
    }
    if (scenario.key === 'seal-launch' || scenario.key === 'seal-write') {
      const witness = acknowledged['sealWitness'];
      const expectedLaunches = scenario.key === 'seal-launch' ? 0 : 1;
      const validWitness =
        typeof witness === 'object' &&
        witness !== null &&
        !Array.isArray(witness) &&
        Object.keys(witness).sort().join(',') ===
          'ackBeforeBodySettled,afterLaunches,afterWrites,beforeLaunches,beforeWrites,kind,refused' &&
        'kind' in witness &&
        witness.kind === (scenario.key === 'seal-launch' ? 'launch' : 'write') &&
        'beforeLaunches' in witness &&
        witness.beforeLaunches === expectedLaunches &&
        'afterLaunches' in witness &&
        witness.afterLaunches === expectedLaunches &&
        'beforeWrites' in witness &&
        witness.beforeWrites === 0 &&
        'afterWrites' in witness &&
        witness.afterWrites === 0 &&
        'refused' in witness &&
        witness.refused === true &&
        'ackBeforeBodySettled' in witness &&
        witness.ackBeforeBodySettled === false;
      if (!validWitness)
        parentFailure = new Error('post-seal actual launch/write refusal witness differs');
      if (
        scenario.key === 'seal-write' &&
        typeof child === 'object' &&
        child !== null &&
        'stdin' in child
      ) {
        const input = child.stdin;
        if (
          typeof input !== 'object' ||
          input === null ||
          !('writes' in input) ||
          input.writes !== 0 ||
          !('refusedAfterSeal' in input) ||
          input.refusedAfterSeal !== 1
        )
          parentFailure = new Error('post-seal physical write count differs');
      }
    }
    await wait(send('retire'), 500).catch(async () => {
      await hold('retirement-control-unobserved');
    });
    outer.stdin.end();
    await wait(Promise.all([exited, closed, stdout, stderr]), 3000).catch(async () => {
      await hold('outer-resource-settlement-unavailable');
    });
    if (
      !facts.stdoutEof ||
      !facts.stderrEof ||
      !facts.inputClosed ||
      !facts.inputFinished ||
      !observed.driverRetiring ||
      facts.exit === undefined ||
      (facts.exit.code === null && facts.exit.signal === null)
    )
      await hold('outer-resource-facts-incomplete');
    console.log(
      JSON.stringify({
        fixture: 'owned-original-retired',
        scenario: scenario.key,
        outer: facts,
        fixtureChild: acknowledged['child'],
        productionCloseReturned: acknowledged['productionCloseReturned'],
        failure: acknowledged['failure'],
        unexpectedFailures: finalDriverFrame?.['unexpectedFailures'],
        expectedStop: finalDriverFrame?.['expectedStop'],
        productionOutcome: acknowledged['productionOutcome'],
        sealWitness: acknowledged['sealWitness'],
      }),
    );
    if (facts.errors.length > 0) parentFailure = new Error('outer fixture emitted an error');
    releaseOwner();
    authorityFixtureOwners.delete(ownership);
  }
  if (parentFailure !== undefined)
    throw parentFailure instanceof Error
      ? parentFailure
      : new Error('fixture driver failed', { cause: parentFailure });
  const acknowledged = frame ?? (await hold('completed-fixture-ACK-unavailable'));
  expect(stderrBytes).toBe(0);
  const final = finalDriverFrame ?? (await hold('final-driver-frame-missing'));
  expect(acknowledged['unexpectedFailures']).toEqual([]);
  expect(final['unexpectedFailures']).toEqual([]);
  expect(acknowledged['failure']).toBeNull();
  expect(final['failure']).toBeNull();
  if (scenario.expectedFailure === undefined) {
    expect(acknowledged['expectedStop']).toBeNull();
    expect(final['expectedStop']).toBeNull();
    expect(facts.exit?.code).toBe(0);
  } else {
    const expected = readClosed(final['expectedStop'], ['kind', 'check', 'name']);
    expect(expected['kind']).toBe('expected-stop');
    expect(expected['check']).toBe(scenario.expectedFailure);
    expect(acknowledged['expectedStop']).toEqual(final['expectedStop']);
    expect(facts.exit?.code).toBe(1);
  }
}

for (const scenario of authorityFixtureCases) {
  test(
    scenario.name,
    async () => {
      await runAuthorityFixture(scenario);
    },
    30000,
  );
}
