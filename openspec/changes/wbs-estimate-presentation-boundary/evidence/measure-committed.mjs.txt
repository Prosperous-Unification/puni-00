import assert from 'node:assert/strict';
import { readCandidate } from '/tmp/wbs-f7-estimates-oct17/apps/twilight-structure/twilight-burokrat/cli/src/inventory/read-candidate.ts';
import { measureSizes } from '/tmp/wbs-f7-estimates-oct17/apps/twilight-structure/twilight-burokrat/cli/src/rules/size-ratchet.ts';
const [repository, baseRevision, candidateRevision, policyPath] = Bun.argv.slice(2);
assert(repository && baseRevision && candidateRevision && policyPath, 'repository, base, candidate, policy required');
const policy = await Bun.file(policyPath).json();
const measure = (revision) => {
  const snapshot = readCandidate(repository, {kind:'committed', revision});
  const outcome = measureSizes(repository, snapshot.entries, policy.sizeCeilings);
  assert(outcome.ok, 'production size evaluation must succeed');
  return {selection:snapshot.selection, outcome};
};
const base = measure(baseRevision);
const candidate = measure(candidateRevision);
const selected = (finding) => finding.path === 'apps/wbs/fe-01/src/components/wbs/plan-columns/estimates.tsx';
console.log(JSON.stringify({base,candidate},null,2));
assert.equal(base.outcome.report.filter(selected).length,1,'base must report selected field once');
// Proof: committed oversized restoration0292ec456 made this fail with selected
// estimates.tsx finding remains; exact good192540c317 passed on restoration.
assert.equal(candidate.outcome.report.filter(selected).length,0,'selected estimates.tsx finding remains');
assert.deepEqual(candidate.outcome.report,base.outcome.report.filter((finding)=>!selected(finding)),'unrelated findings changed');
console.error(`Exact production measureSizes: only selected finding removed (${base.outcome.report.length} -> ${candidate.outcome.report.length}).`);
