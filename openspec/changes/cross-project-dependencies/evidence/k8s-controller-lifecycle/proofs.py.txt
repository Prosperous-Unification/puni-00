def change(before,after,count=1):
 def apply(text):
  if text.count(before)<count:raise Exception('missing mutation anchor '+before)
  return text.replace(before,after,count)
 return apply

def regex(pattern,replacement):
 def apply(text):
  changed,count=re.subn(pattern,replacement,text,count=1,flags=re.S)
  if count!=1:raise Exception('missing regex anchor '+pattern)
  return changed
 return apply
