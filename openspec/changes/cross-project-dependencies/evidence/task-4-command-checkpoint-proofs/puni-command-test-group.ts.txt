import{readFileSync,writeFileSync}from'node:fs';import{spawnSync}from'node:child_process';
const paths:string[]=JSON.parse(readFileSync('/tmp/puni-command-integration-paths.json','utf8'));const group=process.argv[2];let selected:string[],extra:string[]=[];
switch(group){case'core':selected=paths.filter(p=>p.endsWith('.test.ts')&&(p.startsWith('libs/wbs/application/core/')||p.startsWith('libs/wbs/domain/contracts/')));break;
case'http':selected=['command-current-authority','command-organization','directory-command-organization','step-allowance-edit','step','typed-dependency-commands','typed-dependency-copy','typed-dependency-graph','typed-dependency-lifecycle','undo'].map(name=>'apps/wbs/be-01/src/controller/'+name+'.controller.db.test.ts');break;
case'sqlite':selected=paths.filter(p=>p.startsWith('libs/wbs/adapters/store-sqlite/')&&p.endsWith('.test.ts'));break;
case'services':selected=['apps/wbs/be-01/src/services.db.test.ts'];extra=['-t','command|batch|undo|redo|typed|dependency|step|directory'];break;
default:throw new Error('unknown test group');}
writeFileSync('/tmp/puni-command-test-'+group+'-paths.json',JSON.stringify(selected,null,2)+'\n');const execution=spawnSync('bun',['test',...selected.map(p=>'./'+p),...extra,'--timeout=10000'],{encoding:'utf8',timeout:360000,maxBuffer:16000000});const log='/tmp/puni-command-integration-test-'+group+'.log';writeFileSync(log,String(execution.stdout)+String(execution.stderr));writeFileSync('/tmp/puni-command-integration-test-'+group+'.json',JSON.stringify({group,selected,extra,status:execution.status,signal:execution.signal,error:execution.error?.message,log},null,2)+'\n');process.stdout.write(group+': '+execution.status+'\n');process.exit(execution.status??1);
