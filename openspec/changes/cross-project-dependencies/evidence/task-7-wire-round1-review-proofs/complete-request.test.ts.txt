const a = sliceKey('a', null);
const b = sliceKey('b', null);
const c = sliceKey('c', null);

function stage(type: 'FS' | 'SS' | 'FF' = 'FS'): ConnectedOptimizationStage {
  return {
    kind: 'connected-stage',
    projectId: 'B',
    mode: 'isolated',
    originOrdinal: 0,
    projects: [
      { projectId: 'A', rank: 0, anchorOrdinal: 0 },
      { projectId: 'B', rank: 1, anchorOrdinal: 0 },
      { projectId: 'C', rank: 2, anchorOrdinal: 0 },
    ],
    input: {
      rows: ['a', 'b', 'c'].map((id, position) => ({
        id,
        parentId: null,
        position,
        priority: 0,
        frozenNumber: null,
      })),
      slices: [
        { workItemId: 'a', stepId: null, days: 0.2, width: 1, personId: null, poolIds: [] },
        { workItemId: 'b', stepId: null, days: 1, width: 1, personId: null, poolIds: [] },
        { workItemId: 'c', stepId: null, days: 1, width: 1, personId: null, poolIds: [] },
      ],
      graph: {
        nodes: ['a', 'b', 'c'].map((workItemId) => ({
          kind: 'boundary' as const,
          workItemId,
          at: 0,
        })),
        edges: [
          {
            predecessor: { leafId: 'a', at: 0 },
            successor: { leafId: 'b', at: 0 },
            provenance: 'authored',
            relationshipId: 'ab',
            type,
          },
          {
            predecessor: { leafId: 'b', at: 0 },
            successor: { leafId: 'c', at: 0 },
            provenance: 'authored',
            relationshipId: 'bc',
            type,
          },
        ],
      },
      notBefore: new Map(),
      deadlines: new Map(),
      poolSizes: new Map(),
      resourceCapacities: new Map(),
      elsewhere: new Map(),
      order: {
        kind: 'connected',
        projectIdByWorkItem: new Map([
          ['a', 'A'],
          ['b', 'B'],
          ['c', 'C'],
        ]),
        rankByProject: new Map([
          ['A', 0],
          ['B', 1],
          ['C', 2],
        ]),
      },
    },
    relationships: [
      {
        id: 'ab',
        type,
        predecessor: { scope: 'whole', projectId: 'A', workItemId: 'a' },
        successor: { scope: 'whole', projectId: 'B', workItemId: 'b' },
      },
      {
        id: 'bc',
        type,
        predecessor: { scope: 'whole', projectId: 'B', workItemId: 'b' },
        successor: { scope: 'whole', projectId: 'C', workItemId: 'c' },
      },
    ],
    starts: new Map([
      [a, 0],
      [b, 0.2],
      [c, 1.2],
    ]),
    finishes: new Map([
      [a, 0.2],
      [b, 1.2],
      [c, 2.2],
    ]),
  };
}


function prepare(candidate = stage(), budgetMs = 1000) {
 return buildConnectedSolverRequestPair(candidate, {inputHash:'input', questionHashes:{pri:'question-pri',time:'question-time'}}, {solverVersion:'0.4.0',budgetMs});
}

test('a prepared terminal is accepted as a complete request by the installed receiver',async ()=>{
 const candidate=stage();
 Object.assign(candidate,{mode:'shared'});
 Object.assign(candidate.input.slices[2],{days:1e15,personId:'foreign-only-person'});
 (candidate.finishes as Map<string,number>).set(c,1.2+1e15);
 const pair=prepare(candidate);
 expect(pair.pri.kind).toBe('boundary-infeasible');
 if(pair.pri.kind!=='boundary-infeasible') throw new Error('setup requires retained terminal');
 expect(pair.pri.request.elsewhere).toEqual({});
 await Bun.write('/tmp/puni-round1-review-long-foreign-request.json',JSON.stringify(pair.pri.request));
 const child=Bun.spawnSync(['/tmp/puni-cross-solver-wire/.venv-solver/bin/wbs-solver'],{stdin:Buffer.from(JSON.stringify(pair.pri.request))});
 await Bun.write('/tmp/puni-round1-review-long-foreign-cli.json',JSON.stringify({exitCode:child.exitCode,stdout:child.stdout.toString(),stderr:child.stderr.toString()}));
 console.log('terminal',pair.pri.evidence,'retained occupancy',pair.pri.request.scope.occupancy,'CLI',child.exitCode,child.stderr.toString());
 expect(child.exitCode).toBe(0);
});
