from pathlib import Path
import subprocess
root = Path('/tmp/puni-external-command-contracts')
fields = root / 'libs/wbs/application/core/src/service/command-fields.ts'
definitions = root / 'libs/wbs/domain/contracts/src/commands/definitions.ts'
shape = root / 'libs/wbs/domain/contracts/src/http/plan-command-shapes.ts'
client = root / 'apps/wbs/mcp-01/src/wbs-client.ts'
boot = 'apps/wbs/mcp-01/src/organization-delegation.boot.test.ts'
document = 'apps/wbs/mcp-01/src/generated-document.test.ts'

def replace_once(source, before, after):
    if source.count(before) != 1: raise RuntimeError(f'Expected unique mutation: {before}')
    return source.replace(before, after)

def widen(source, kind):
    start = source.index(f'  {kind}: defineCommand(')
    end = source.index("      type: \"'FS' | 'SS' | 'FF'\",", start)
    return source[:end] + source[end:].replace("      type: \"'FS' | 'SS' | 'FF'\",", "      type: 'string',", 1)

def remove_guard(source):
    start = source.index('  if (\n    projectId !== undefined &&\n    [projectId, workItemId, stepId].some(')
    end = source.index("    throw new CommandNormalizationError('invalid_typed_endpoint');", start)
    return source[:start] + source[end + len("    throw new CommandNormalizationError('invalid_typed_endpoint');"):]

faults = [
    ('explicit-id', fields, remove_guard, boot),
    ('add-sf', definitions, lambda source: widen(source, 'addTypedDependency'), boot),
    ('update-sf', definitions, lambda source: widen(source, 'updateTypedDependency'), boot),
    ('nested-extra', shape, lambda source: replace_once(source, 'export const planCommandsBody = requestSchema(commandsBody);', "export const planCommandsBody = requestSchema(commandsBody, { undeclaredKeys: 'delete' });"), boot),
    ('explicit-alias', fields, lambda source: replace_once(source, "  if (projectId !== undefined && workItemRef !== undefined)\n    throw new CommandNormalizationError('invalid_typed_endpoint');", ''), boot),
    ('serialized-project', client, lambda source: replace_once(source, 'body: JSON.stringify(body)', "body: JSON.stringify(body, (key, value) => key === 'projectId' ? undefined : value)"), boot),
    ('published-project', definitions, lambda source: replace_once(source, "const dependencyEndpoint = type({\n  'projectId?': 'string',", 'const dependencyEndpoint = type({'), document),
]
for name, path, mutate, suite in faults:
    original = path.read_text()
    try:
        path.write_text(mutate(original))
        with open(f'/tmp/puni-parity-fault-{name}.log', 'w') as log:
            execution = subprocess.run(['bun', 'test', suite], cwd=root, stdout=log, stderr=subprocess.STDOUT, timeout=180)
        lines = Path(f'/tmp/puni-parity-fault-{name}.log').read_text().splitlines()
        failures = [line for line in lines if line.startswith('(fail)')]
        print(f'{name}: exit {execution.returncode}; ' + '; '.join(failures), flush=True)
        if execution.returncode == 0 or not failures: raise RuntimeError(f'Fault {name} did not produce an observed test assertion failure')
    finally:
        path.write_text(original)
