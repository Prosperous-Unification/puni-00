checks=[

 ('runtime-lock',lib,lambda s:s.replace("join(directory, 'solver-runtime.lock')", "join(directory, 'wrong-runtime.lock')"), 'actual CLI retains runtime after readiness'),
 ('runtime-strict',lib,lambda s:s.replace("import { withCanonicalLock }", "import { withLock as withCanonicalLock }"), 'runtime refuses symlink lock'),
 ('owner-lock',owner,lambda s:s.replace("join(ownership.directory, 'solver-runtime.lock')", "join(ownership.directory, 'wrong-runtime.lock')"), 'paused actual startup'),
 ('owner-entry',owner,lambda s:s[:s.index('export async function withOwnerSolverRuntime')]+s[s.index('export async function withOwnerSolverRuntime'):].replace('await assertDependencyOwner(owner, layout);','',2), 'runtime owner checks retained publication before'),
 ('runtime-marker',lib,lambda s:s.replace('await assertNoEdgeOwner(directory);',''), 'actual CLI edge-owner'),

evidence=root/'openspec/changes/cross-project-dependencies/evidence/solver-runtime-admission'
for name,path,mutate,pattern in checks:
 original=path.read_text(); changed=mutate(original)
 if original==changed: raise RuntimeError('mutation did not apply '+name)
 try:
  path.write_text(changed)
  command=['bun','test','tools/tool-remote-scripts/src/lib/solver-runtime-admission.test.ts','tools/tool-remote-scripts/src/solver-runtime-admission.proc.test.ts','--preload','./tools/test/scratch/preload.ts','--timeout=10000','-t',pattern]

  (evidence/(name+'.log.txt')).write_text(' '.join(command)+'\n'+completed.stdout+'\nexit '+str(completed.returncode)+'\n')
  print(name,completed.returncode,completed.stdout[-180:],flush=True)
  if completed.returncode==0: raise RuntimeError('mutation survived '+name)
 finally: path.write_text(original)
