function digest(bytes:Uint8Array){return createHash('sha256').update(bytes).digest('hex');}
function filePin(path:string){const bytes=readFileSync(path);return{path,sha256:digest(bytes),bytes:bytes.length};}
function artifactPin(path:string){const pin=filePin(path);return{file:basename(path),sha256:pin.sha256,bytes:pin.bytes};}

function readLedger(db:Drizzle){
  const candidates=new Map(readMigrationFolders(MIGRATIONS_FOLDER).map(folder=>[folder.name,folder.hash]));
  const names=new Set<string>();
  return db.all<{name:unknown;hash:unknown}>(sql`SELECT name,hash FROM __drizzle_migrations ORDER BY name COLLATE BINARY`).map(row=>{
    if(typeof row.name!=='string'||row.name===''||typeof row.hash!=='string'||!(/^[0-9a-f]{64}$/).test(row.hash)||candidates.get(row.name)!==row.hash||names.has(row.name))throw new Error('genuine fixture ledger is malformed, duplicated or edited');
    names.add(row.name);return{name:row.name,hash:row.hash};
  });
}

function captureStore(db:Drizzle){
  const schema=db.all(sql`SELECT type,name,tbl_name,sql FROM sqlite_master ORDER BY type COLLATE BINARY,name COLLATE BINARY`);
  const tables=db.all<{name:unknown}>(sql`SELECT name FROM sqlite_master WHERE type='table' ORDER BY name COLLATE BINARY`).map(({name})=>{
    if(typeof name!=='string'||name==='')throw new Error('frozen fixture table name is malformed');
    const rows=db.all<Record<string,unknown>>(sql.raw('SELECT * FROM "'+name.replaceAll('"','""')+'"'));
    rows.sort((left,right)=>Buffer.compare(Buffer.from(JSON.stringify(left)),Buffer.from(JSON.stringify(right))));
    return{name,rows};
  });
  return{schema,tables,ledger:readLedger(db)};
}
